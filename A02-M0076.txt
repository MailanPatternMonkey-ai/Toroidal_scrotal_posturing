import json
import math
import os
import sys

class ParallelLedgerValidator:
    """
    Parallel Axis Ledger Validator & Integrity Engine.
    Processes append-only .jsonl logs to audit structural compliance,
    detect illegal state leaks, and isolate cross-stream contamination.
    """
    def __init__(self, filename="parallel_axis_ledger.jsonl"):
        self.filename = filename
        
        # Invariant UI Color Anchors
        self.BOLD = "\033[1m"
        self.CYAN = "\033[36m"
        self.GOLD = "\033[33m"
        self.RED = "\033[31m"
        self.GREEN = "\033[32m"
        self.RESET = "\033[0m"

    def _is_prime(self, n: int) -> bool:
        if n < 2: return False
        for i in range(2, int(math.isqrt(n)) + 1):
            if n % i == 0: return False
        return True

    def audit_ledger_integrity(self):
        """Runs a line-by-line structural audit over the stored telemetry rows."""
        print(f"\n{self.BOLD}{self.CYAN}[AUDIT INITIALIZED] SCANNING LEDGER FILE FOR HARD LEAKS...")
        print(f"[AUDIT TARGET] source_file: {self.filename}")
        print(f"+" + "="*95 + "+")

        if not os.path.exists(self.filename):
            print(f"{self.BOLD}{self.RED}[FATAL ERROR] Target ledger file not found on disk.{self.RESET}\n")
            return False

        total_rows = 0
        structural_failures = 0
        leakage_warnings = 0
        critical_null_records = 0

        with open(self.filename, "r", encoding="utf-8") as f:
            for line_idx, line in enumerate(f, 1):
                total_rows += 1
                try:
                    record = json.loads(line.strip())
                except json.JSONDecodeError as err:
                    print(f"{self.BOLD}{self.RED}[ROW ERROR] Row {line_idx:04d} -> Corrupted JSON block: {err}{self.RESET}")
                    structural_failures += 1
                    continue

                # 1. Structural Schema Verification
                if "clocks" not in record or "streams" not in record:
                    print(f"{self.BOLD}{self.RED}[SCHEMA FAILURE] Row {line_idx:04d} -> Missing vital core namespace maps.{self.RESET}")
                    structural_failures += 1
                    continue

                clocks = record["clocks"]
                streams = record["streams"]
                
                # Check for baseline clock desynchronization or tampering
                dt_slip = clocks.get("absolute_substrate_slip", 0.0)
                tg = clocks.get("lattice_clock_tg", 0.0)
                te = clocks.get("execution_clock_te", 0.0)
                
                if not math.isclose(abs(te - tg), dt_slip, abs_tol=1e-5):
                    print(f"{self.BOLD}{self.RED}[CLOCK FRAUD] Row {line_idx:04d} -> Absolute slip value modified post-run.{self.RESET}")
                    structural_failures += 1

                # 2. Parallel Stream Validation & Cross-Contamination Inspection
                x_node = streams["axis_x"]["current_node"]
                y_node = streams["axis_y"]["current_node"]
                z_node = streams["axis_z"]["current_node"]
                
                # Cross-Contamination Test: Check if distinct trajectories became locked or cloned illegally
                if x_node == y_node and x_node == z_node and not self._is_prime(x_node):
                    print(f"{self.BOLD}{self.GOLD}[CONTAMINATION WARNING] Row {line_idx:04d} -> Channels [X,Y,Z] collapsed into same composite node: {x_node}{self.RESET}")
                    leakage_warnings += 1

                # 3. Parity Invariant Verification
                for ax in ["axis_x", "axis_y", "axis_z"]:
                    stream_data = streams[ax]
                    node_id = stream_data["current_node"]
                    expected_q = 1 if self._is_prime(node_id) else (node_id % 3) + 1
                    
                    if stream_data["q_parity"] != expected_q:
                        print(f"{self.BOLD}{self.RED}[PARITY CORRUPTION] Row {line_idx:04d} -> Axis {ax.upper()} Node {node_id} mapped to invalid Parity Quotient: {stream_data['q_parity']}{self.RESET}")
                        structural_failures += 1

                    # Track occurrences of isolated Critical Null flags saved to disk
                    if stream_data["catch_witness"] > 0.42:
                        critical_null_records += 1

        # Render Summary Report Matrix
        print(f"+" + "="*95 + "+")
        print(f"{self.BOLD}{self.CYAN}[AUDIT COMPLETE] RECONCILIATION SUMMARY RECORDED:")
        print(f" -> Total Telemetry Rows Scanned  : {total_rows}")
        print(f" -> Invariant Structural Failures : {self.RED if structural_failures > 0 else self.GREEN}{structural_failures}{self.RESET}")
        print(f" -> Cross-Stream Contamination Warnings : {self.GOLD if leakage_warnings > 0 else self.GREEN}{leakage_warnings}{self.RESET}")
        print(f" -> Stamped Critical Null Overflows : {self.CYAN}{critical_null_records}{self.RESET}")
        print(f"+" + "="*95 + "+")

        if structural_failures > 0:
            print(f"{self.BOLD}{self.RED}[STATUS] LEDGER REJECTED. SCHEMA OR COCYCLE INVARIANTS BROKEN. ❌{self.RESET}\n")
            return False
        else:
            print(f"{self.BOLD}{self.GREEN}[STATUS] LEDGER VERIFIED CLEAN. LINEAGE HOLDS PROVENANCE. ✊{self.RESET}\n")
            return True

if __name__ == "__main__":
    # Ingest the log file for validation review
    validator = ParallelLedgerValidator(filename="parallel_axis_ledger.jsonl")
    validator.audit_ledger_integrity()
