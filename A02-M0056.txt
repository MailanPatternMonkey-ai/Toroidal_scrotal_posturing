class MultiAxisLedgerSplitter:
    """
    Module 12 / A.15 Multi-Axis Compliance Engine.
    Simultaneously drives and tracks independent parallel execution paths 
    for X, Y, and Z spatial dimensions, logging them into synchronized 
    multi-column array streams without cross-contamination.
    """
    def __init__(self, filename="multi_axis_cocycle_ledger.jsonl"):
        self.filename = filename

    def log_parallel_streams(self, step: int, active_nodes: dict, clocks: dict, 
                             witness_deltas: dict, aliased_flags: dict, status_signatures: dict):
        """
        Stamps a completely parallelized, multi-column structural record of 
        all three geometric axes into the immutable tracking pipeline.
        """
        ms_timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S.%f")[:-3]
        
        # Parallel Matrix Row Compilation
        parallel_record = {
            "source_timestamp": ms_timestamp,
            "step_index": step,
            "clocks": {
                "geometric_lattice_time_tg": round(clocks["tg"], 6),
                "execution_processing_time_te": round(clocks["te"], 6),
                "absolute_substrate_slip_dt": round(abs(clocks["te"] - clocks["tg"]), 6)
            },
            "axis_streams": {
                "X_AXIS": {
                    "node_coordinate": active_nodes["x"],
                    "catch_witness_delta": round(witness_deltas["x"], 6),
                    "parity_quotient_q": 1 if self._is_prime(active_nodes["x"]) else (active_nodes["x"] % 3) + 1,
                    "frequency_aliased": aliased_flags["x"],
                    "status_signature": status_signatures["x"]
                },
                "Y_AXIS": {
                    "node_coordinate": active_nodes["y"],
                    "catch_witness_delta": round(witness_deltas["y"], 6),
                    "parity_quotient_q": 1 if self._is_prime(active_nodes["y"]) else (active_nodes["y"] % 3) + 1,
                    "frequency_aliased": aliased_flags["y"],
                    "status_signature": status_signatures["y"]
                },
                "Z_AXIS": {
                    "node_coordinate": active_nodes["z"],
                    "catch_witness_delta": round(witness_deltas["z"], 6),
                    "parity_quotient_q": 1 if self._is_prime(active_nodes["z"]) else (active_nodes["z"] % 3) + 1,
                    "frequency_aliased": aliased_flags["z"],
                    "status_signature": status_signatures["z"]
                }
            },
            "relation_firewall": {
                "mapping_rules": "EXACT_QUOTIENT_PER_AXIS",
                "co-primary_tracking": "THREE_AXIS_FIXED_HOLONOMY_ENSEMBLE"
            }
        }

        with open(self.filename, "a", encoding="utf-8") as f:
            f.write(json.dumps(parallel_record) + "\n")

    def _is_prime(self, n: int) -> bool:
        if n < 2: return False
        for i in range(2, int(math.isqrt(n)) + 1):
            if n % i == 0: return False
        return True

    def _is_prime(self, n: int) -> bool:
        if n < 2: return False
        for i in range(2, int(math.isqrt(n)) + 1):
            if n % i == 0: return False
        return True

    def get_allowed_moves(self, current: int):
        moves = []
        if current + 1 < self.max_range: moves.append(current + 1)
        if current - 1 > 0: moves.append(current - 1)
        if current * 2 < self.max_range: moves.append(current * 2)
        if current % 2 == 0 and current // 2 > 0: moves.append(current // 2)
        return moves

    def calculate_node_friction(self, node: int, axis_label: str) -> int:
        base_cost = 0 if self._is_prime(node) else sum(1 for i in range(1, node + 1) if node % i == 0) * 15
        aliasing_incidents = self.sector_aliasing_history[axis_label].get(node, 0)
        if aliasing_incidents >= self.aliasing_threshold_limit:
            return base_cost + (500 * aliasing_incidents)
        return base_cost + (aliasing_incidents * 50)

    def find_optimal_path(self, start: int, target: int, axis_label: str) -> list:
        if not self._is_prime(target): return []
        priority_queue = [(0, start, [start])]
        min_costs = {start: 0}

        while priority_queue:
            current_aggro, current, path = heapq.heappop(priority_queue)
            if current == target: return path
            if current_aggro > min_costs.get(current, float('inf')): continue
            for next_node in self.get_allowed_moves(current):
                step_friction = self.calculate_node_friction(next_node, axis_label)
                total_next_aggro = current_aggro + step_friction
                if total_next_aggro < min_costs.get(next_node, float('inf')):
                    min_costs[next_node] = total_next_aggro
                    heapq.heappush(priority_queue, (total_next_aggro, next_node, path + [next_node]))
        return []
