\[
\pi\circ\Psi=\bar\Psi\circ\pi.
\]

Equivalently,

\[
\pi(x)=\pi(y)\Longrightarrow\pi(\Psi x)=\pi(\Psi y).
\]

\[
P_\pi(x,q')
=
\sum_{z:\pi(z)=q'}P(x,z).

\[
\boxed{
\pi(x)=\pi(y)
\Longrightarrow
P_\pi(x,q')=P_\pi(y,q')
\quad\text{for every }q'.
