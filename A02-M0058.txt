import sys
import time
import math

class TerminalClockMatrix:
    """
    Module 6 Execution Wrapper.
    Renders an active, live-updating terminal dashboard tracking the drift
    between System Processing Time (te) and Geometric Lattice Time (tg).
    """
    def __init__(self, lattice_period=0.25, target_sector=113):
        self.Tg = lattice_period
        self.target_sector = target_sector
        
        # ANSI Escape Codes for crisp terminal UI rendering
        self.BOLD = "\033[1m"
        self.CYAN = "\033[36m"
        self.GOLD = "\033[33m"
        self.RED = "\033[31m"
        self.GREEN = "\033[32m"
        self.RESET = "\033[0m"
        self.CLEAR_SCREEN = "\033[2J\033[H"
        self.HIDE_CURSOR = "\033[?25l"
        self.SHOW_CURSOR = "\033[?25h"

    def _is_prime(self, n: int) -> bool:
        if n < 2: return False
        for i in range(2, int(math.isqrt(n)) + 1):
            if n % i == 0: return False
        return True

    def run_live_matrix(self, ticks=40):
        sys.stdout.write(self.HIDE_CURSOR)
        sys.stdout.flush()
        
        start_physical_time = time.time()
        geometric_lattice_time = 0.0
        accumulated_drift = 0.0

        try:
            for step in range(1, ticks + 1):
                # 1. Update the Dual Clocks
                geometric_lattice_time += self.Tg
                
                # Introduce artificial environmental delay
                # Prime targets trigger 0 processing lag; composite steps create jitter
                if self._is_prime(step):
                    time.sleep(self.Tg)
                else:
                    factors = sum(1 for i in range(1, step + 1) if step % i == 0)
                    jitter = (factors * 0.012) + (step * 0.001)
                    time.sleep(self.Tg + jitter)

                execution_processing_time = time.time() - start_physical_time
                
                # 2. Calculate Phase Slip and Catch Witness Metrics
                # Phase equation: g(t) = frac(gs - t/Tg)
                q_quotient = 1 if self._is_prime(step) else (step % 4) + 1
                theta_n = (step * 0.1337) % 1.0
                
                gs = step * 0.5
                raw_phase = gs - (geometric_lattice_time / (self.Tg / q_quotient))
                lattice_phase = raw_phase - math.floor(raw_phase)
                
                phase_divergence = abs(theta_n - lattice_phase)
                delta_catch = min(phase_divergence, 1.0 - phase_divergence) + ((execution_processing_time - geometric_lattice_time) * 0.05)
                delta_catch = max(0.0, delta_catch)

                # 3. Determine Bank Verdict & System Vulnerability Thresholds
                if delta_catch > 0.42:
                    verdict_color = self.RED
                    verdict_text = "OLD BANK VERDICT: NULL // CRITICAL TELEMETRY DRIFT"
                    status_text = "🔒 PLATFORM ANCHOR DEPLOYED — REFUSING MARKOV CLOSURE"
                elif delta_catch > 0.25:
                    verdict_color = self.GOLD
                    verdict_text = "OLD BANK VERDICT: VOLATILE // TRACKING DELTA LAG"
                    status_text = "⚠️  CAUTION: HEAVY DRAG DETECTED UPSTREAM"
                else:
                    verdict_color = self.GREEN
                    verdict_text = "OLD BANK VERDICT: STABLE"
                    status_text = "🟢 DUAL-CLOCK SLIPSTREAM SYNCHRONIZED"

                # 4. Render UI Frame Buffer directly to Terminal Output
                frame = f"""{self.CLEAR_SCREEN}{self.CYAN}+=============================================================================+

| {self.BOLD}{self.GOLD}     TWO-CLOCK PROSPECTIVE MATRIX // LATTICE DRIFT ENGINE (v0.4-RC1)    {self.RESET}{self.CYAN} |
+=============================================================================+

|                                                                             |
|  {self.BOLD}SYSTEM TIME PARITY CHECK:{self.RESET}{self.CYAN}                                                  |
|  [+] Step Index : {step:03d} / {ticks:03d}                                               |
|  [+] Target Node: Sector {self.target_sector:<3} (Irreducible Prime Anchor)                    |
|                                                                             |
|  {self.BOLD}CHRONO METRICS:{self.RESET}{self.CYAN}                                                           |
|  -> Geometric Lattice Time (tg) : {self.BOLD}{self.CYAN}{geometric_lattice_time:.4f} s{self.RESET}{self.CYAN}                              |
|  -> Physical Execution Time (te): {self.BOLD}{self.CYAN}{execution_processing_time:.4f} s{self.RESET}{self.CYAN}                              |
|  -> Absolute Substrate Slip (Δt): {self.BOLD}{self.RED}{abs(execution_processing_time - geometric_lattice_time):.4f} s{self.RESET}{self.CYAN}                              |
|                                                                             |
|  {self.BOLD}PHASE RESOLUTION:{self.RESET}{self.CYAN}                                                         |
|  -> Catch Witness (δ_catch)     : {verdict_color}{delta_catch:.6f}{self.RESET}{self.CYAN}                             |
|  -> Unlabelled Matrix Period     : {self.Tg / q_quotient:.4f} s (Tg/q Frequency Aliasing)       |
|                                                                             |
|  {self.BOLD}SECURITY ALIGNMENT:{self.RESET}{self.CYAN}                                                       |
|  -> STATUS : {self.BOLD}{verdict_color}{verdict_text}{self.RESET}{self.CYAN}             |
|  -> ACTION : {self.BOLD}{status_text}{self.RESET}{self.CYAN}             |
|                                                                             |
+=============================================================================+

|  {self.BOLD}{self.RED}  UNION 113 RULES ENFORCED • DIGNITY SECURED • FREED FROM MASTERS & BULLSHIT {self.RESET}{self.CYAN} |
+=============================================================================+{self.RESET}
"""
                sys.stdout.write(frame)
                sys.stdout.flush()

        except KeyboardInterrupt:
            pass
        finally:
            sys.stdout.write(self.SHOW_CURSOR + "\n")
            sys.stdout.flush()

if __name__ == "__main__":
    # Initialize the dual-clock dashboard mapping to the 113 flag coordinates
    matrix = TerminalClockMatrix(lattice_period=0.15, target_sector=113)
    matrix.run_live_matrix(ticks=40)
